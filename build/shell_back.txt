function shuffle(a,seed){
  const arr=a.slice();
  if(seed==null){for(let i=arr.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[arr[i],arr[j]]=[arr[j],arr[i]];}return arr;}
  let s=seed>>>0;const rnd=()=>{s=(s*1664525+1013904223)>>>0;return s/4294967296;};
  for(let i=arr.length-1;i>0;i--){const j=Math.floor(rnd()*(i+1));[arr[i],arr[j]]=[arr[j],arr[i]];}
  return arr;
}

function ensureTimer(){

  const tick=()=>{
    const el=document.getElementById('timer');
    if(!el){stopTimer();return;}

    const m=Math.floor(left/60000),s=Math.floor(left%60000/1000);
    el.textContent='⏱ '+m+':'+String(s).padStart(2,'0');
  };
  tick();
  timerId=setInterval(tick,1000);
}

/* ---------- 组卷 ---------- */
function mkItem(q,uid,cid,qi,chn){
  return Object.assign({t:qKind(q),lv:q.lv||1,pt:q.pt||1,sec:q.sec||null,chid:uid+'_'+cid,chn,qi},q);
}
function poolOfChapter(uid,cid){
  const u=DATA.units[uid],c=u.chapters[cid];
  return c.quiz.map((q,qi)=>mkItem(q,u.id,c.id,qi,u.emoji+' '+c.name));
}
function poolOfAll(){
  const r=[];
  DATA.units.forEach((u,ui)=>u.chapters.forEach((c,ci)=>r.push.apply(r,poolOfChapter(ui,ci))));
  return r;
}
function startQuiz(mode,extra){
  extra=extra||{};
  const qz={i:0,got:0,score:0,objScore:0,objN:0,max:0,wrong:[],done:false,mode:mode,deadline:null,work:{}};
  if(mode==='wrong'){
    const wb=lsGet('wrongbook',[]);
    if(!wb.length){return;}
    qz.list=wb.map(w=>Object.assign({},w,{t:w.t||'choice'}));
    qz.title='错题重做';
  }else if(mode==='daily'){
    qz.list=shuffle(poolOfAll(),hashStr(ymd())).slice(0,10);
    qz.title='每日挑战';
    qz.deadline=Date.now()+5*60000;
  }else if(mode==='gate'){
    const u=DATA.units[state.unit],c=u.chapters[state.ch];
    const si=extra.si;const sid=c.sections[si].id||('s'+(si+1));
    qz.list=poolOfChapter(state.unit,state.ch).filter(it=>it.sec===sid);
    qz.title='闯关 · '+c.sections[si].name;
    qz.gateSi=si;qz.gateSid=sid;
  }else{
    const u=DATA.units[state.unit],c=u.chapters[state.ch];
    let pool=poolOfChapter(state.unit,state.ch);
    if(mode==='custom'){
      const pc=state.pick;
      pool=pool.filter(it=>((!it.sec||pc.secs.includes(it.sec))&&pc.lvs.includes(it.lv)));
      if(!pool.length){alert('所选范围内没有题目，请调整选择。');return;}
      qz.title='自选练习';
    }else{
      qz.title='本章练习';
    }
    qz.list=shuffle(pool);
  }
  if(!qz.list.length){alert('这个范围内没有题目。');return;}
  qz.list.forEach(it=>{const t=qKind(it);if(t!=='essay'){qz.max+=(it.pt||1);qz.objN++;}else{qz.max+=(it.subs||[]).reduce((a,s)=>a+(s.score||0),0);}});
  state.quiz=qz;state.view='quiz';render();
}

  const i=qz.work.picked.indexOf(d);
  if(i>=0)qz.work.picked.splice(i,1);
  else qz.work.picked.push(d);

/* ---------- 判分 ---------- */
function normAns(s){return String(s).replace(/\s+/g,'').replace(/[，。、；：！？,.;:!?（）()]/g,'').toLowerCase();}
function nearMiss(a,b){
  a=normAns(a);b=normAns(b);
  if(Math.abs(a.length-b.length)>1)return false;
  if(a===b)return true;
  const s=a.length<b.length?a:b,l=a.length<b.length?b:a;
  if(s.length===l.length){let d=0;for(let i=0;i<s.length;i++){if(s[i]!==l[i])d++;}return d<=1;}
  let i=0,j=0,d=0;while(i<s.length&&j<l.length){if(s[i]!==l[j]){d++;j++;if(d>1)return false;}else{i++;j++;}}return d<=1;
}
function submitAns(val){
  const qz=state.quiz;if(qz.done)return;
  const cur=qz.list[qz.i],t=qKind(cur);
  let res=null;
  if(t==='choice'||t==='tf')res=gradeTap(cur,val);
  else if(t==='fill')res=gradeFill(cur,val);
  else if(t==='order')res=gradeOrder(cur,val);
  else if(t==='essay')res=gradeEssay(cur,val);
  afterAnswer(cur,res);
}
function gradeTap(it,oi){
  const t=qKind(it);
  const ok=t==='tf'?((oi===0)===it.ans):(oi===Number(it.ans));
  const rightIdx=t==='tf'?(it.ans?0:1):Number(it.ans);
  const optEls=document.querySelectorAll('.opt');
  optEls.forEach(o=>o.classList.add('dim'));
  const fb=document.getElementById('fb');
  if(ok){
    optEls[oi].classList.add('right');
    fb.className='fb ok';fb.innerHTML='🎉 '+(PRAISE[Math.floor(Math.random()*PRAISE.length)])+'<span class="why">📝 解析：'+esc(it.why)+'</span>';
  }else{
    optEls[oi].classList.add('wrong');optEls[rightIdx].classList.add('right');
    fb.className='fb bad';fb.innerHTML='❌ 答错了<span class="why">📝 解析：'+esc(it.why)+'</span>';
  }
  return {ok:ok,got:ok?(it.pt||1):0,html:fb.innerHTML};
}
function matchFill(s,al){
  s=normAns(s);
  if(s.length<2)return false;  // 少于2个字不判（防虚词/误触）
  return al.some(a=>{
    const na=normAns(a);
    if(s===na)return true;
    // 关键词判定：孩子的答案含标准答案（或其关键词），或反之——有关键词就判对
    return (na.length>=2&&s.includes(na))||(s.length>=2&&na.includes(s));
  });
}
function gradeFill(it,arr){
  const pairs=(it.a||it.ans);
  const okAll=pairs.every((al,k)=>matchFill(arr[k],al));
  const near=pairs.some((al,k)=>!matchFill(arr[k],al)&&al.some(a=>nearMiss(arr[k],a)));
  const ansShow=pairs.map(al=>al[0]).join('、');
  const fb=document.getElementById('fb');
  if(okAll){
    fb.className='fb ok';fb.innerHTML='🎉 填空正确！<span class="why">📝 解析：'+esc(it.why)+'</span>';
  }else{
    fb.className='fb bad';
    fb.innerHTML='❌ 有空格不对<span class="why">📝 答案：'+esc(ansShow)+'｜解析：'+esc(it.why)+'</span>'
      +(near?'<button class="btn2" style="margin-top:8px" onclick="fillOverride()">我写的其实是对的（拼写小误差）</button>':'');
  }
  return {ok:okAll,got:okAll?(it.pt||1):0,html:fb.innerHTML,near:near};
}

function gradeOrder(it,seq){

  const fb=document.getElementById('fb');

  }
  return {ok:ok,got:ok?(it.pt||1):0,html:fb.innerHTML};
}
function gradeEssay(it,ks){
  let got=0,full=true;
  it.subs.forEach((s,si)=>{
    const k=ks[si]||0;got+=Math.round((s.score||0)*k);
    if(k!==1)full=false;
  });
  const fb=document.getElementById('fb');
  const max=(it.subs||[]).reduce((a,s)=>a+(s.score||0),0);
  if(full){
    fb.className='fb ok';fb.innerHTML='🏆 全部要点都答到了，满分 '+max+' 分！<span class="why">📝 解析：'+esc(it.why)+'</span>';
  }else{
    fb.className='fb bad';fb.innerHTML='📝 自评得分 '+got+' / '+max+' 分<span class="why">没答全的要点要记住哦：'+esc(it.why)+'</span>';
  }
  return {ok:full,got:got,html:fb.innerHTML};
}
function afterAnswer(cur,res){
  const qz=state.quiz;
  qz.got+=res.got;
  if(res.ok){qz.score++;if(qKind(cur)!=='essay')qz.objScore++;}
  else if(!res.override)qz.wrong.push(cur);
  statsAdd(res.ok,qKind(cur)==='essay'?res.got:0);
  qz.done=true;
  const nb=document.getElementById('nextbtn');
  nb.style.display='block';
  nb.textContent=(qz.i<qz.list.length-1)?'下一题 →':'查看成绩 🏁';
}

function ansText(w){
  const t=w.t||w.type||'choice';
  if(t==='tf')return w.ans?'✔ 正确':'✘ 错误';
  if(t==='fill')return ((w.a||w.ans||[[]])[0]&&(w.a||w.ans)[0][0])||'见解析';
  if(t==='order')return w.items?w.items.join(' → '):'见解析';
  if(t==='essay')return '见要点（重做时对照）';
  if(w.opts&&w.opts[w.ans]!=null)return optText(w.opts[w.ans]);
  return '';
}
